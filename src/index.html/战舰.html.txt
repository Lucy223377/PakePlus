    if (!Array.isArray(data.cleared)) return { cleared: [] };
    return data;
  } catch (e) {
    return { cleared: [] };
  }
}

function saveProgress(progress) {
  try {
    localStorage.setItem(SAVE_KEY, JSON.stringify(progress));
  } catch (e) {}
}

function markCleared(levelId) {
  try {
    const p = loadProgress();
    if (!p.cleared.includes(levelId)) {
      p.cleared.push(levelId);
      saveProgress(p);
    }
  } catch (e) {}
}

function isCleared(levelId) {
  try { return loadProgress().cleared.includes(levelId); }
  catch (e) { return false; }
}

function isUnlocked(levelId) {
  if (levelId === 1) return true;
  return isCleared(levelId - 1);
}

function resetProgress() {
  try {
    if (!confirm('确定要清空通关进度吗？此操作不可撤销。')) return;
    localStorage.removeItem(SAVE_KEY);
  } catch (e) {}
  renderCampaign();
}

    `;

/* ================================================================
   视野
   ================================================================ */

function lineOfSight(x0, y0, x1, y1) {
  let dx = Math.abs(x1 - x0);
  let dy = Math.abs(y1 - y0);
  let sx = x0 < x1 ? 1 : -1;
  let sy = y0 < y1 ? 1 : -1;
  let err = dx - dy;
  let x = x0, y = y0;
  while (true) {
    if (x === x1 && y === y1) break;
    const e2 = 2 * err;
    if (e2 > -dy) { err -= dy; x += sx; }
    if (e2 < dx)  { err += dx; y += sy; }
    if (x === x1 && y === y1) break;
    if (map[y][x] === 1) return false;
  }
  return true;
}

function computeVisionFrom(unit) {
  const set = new Set();
  const r = unit.vision;
  for (let dy = -r; dy <= r; dy++) {
    for (let dx = -r; dx <= r; dx++) {
      const dist = Math.abs(dx) + Math.abs(dy);
      if (dist > r) continue;
      const x = unit.x + dx;
      const y = unit.y + dy;
      if (x < 0 || x >= SIZE || y < 0 || y >= SIZE) continue;
      if (dx === 0 && dy === 0) { set.add(`${x},${y}`); continue; }
      if (lineOfSight(unit.x, unit.y, x, y)) set.add(`${x},${y}`);
    }
  }
  return set;
}

function recomputeVision() {
  visionSet = new Set();
  for (const u of units) {
    if (u.side !== 'player' || u.hp <= 0) continue;
    for (const k of computeVisionFrom(u)) visionSet.add(k);
  }
}

function isVisible(x, y) { return visionSet.has(`${x},${y}`); }
function isUnitVisibleToPlayer(u) {
  if (u.side === 'player') return true;
  if (u.hp <= 0) return false;
  return isVisible(u.x, u.y);
}

/* ================================================================
   移动 / 攻击 / 放飞
   ================================================================ */

function getCurrentMovePower(unit) {
  if (unit.kind === 'plane') return unit.moveLeft;
  return unit.hasMoved ? 0 : unit.move;
}

function getMoveRange(unit) {
  const power = getCurrentMovePower(unit);
  const visited = new Map();
  visited.set(`${unit.x},${unit.y}`, 0);
  const queue = [{ x: unit.x, y: unit.y, cost: 0 }];
  const isPlane = unit.kind === 'plane';

  while (queue.length > 0) {
    const cur = queue.shift();
    if (cur.cost >= power) continue;
    for (const [dx, dy] of [[1,0],[-1,0],[0,1],[0,-1]]) {
      const nx = cur.x + dx, ny = cur.y + dy;
      if (nx < 0 || nx >= SIZE || ny < 0 || ny >= SIZE) continue;
      if (!isPlane && map[ny][nx] === 1) continue;
      const key = `${nx},${ny}`;
      if (visited.has(key)) continue;
      const occ = getUnitAt(nx, ny);
      if (occ && occ.side !== unit.side) continue;
      visited.set(key, cur.cost + 1);
      queue.push({ x: nx, y: ny, cost: cur.cost + 1 });
    }
  }

  const result = new Set();
  for (const key of visited.keys()) {
    const [x, y] = key.split(',').map(Number);
    if (map[y][x] === 1) continue;
    const occ = getUnitAt(x, y);
    if (occ && occ !== unit) continue;
    result.add(key);
  }
  result.delete(`${unit.x},${unit.y}`);
  return result;
}

function refreshAttackable(unit) {
  attackableSet.clear();
  if (!unit) return;
  if (unit.kind === 'plane' && unit.acted) return;
  if (unit.kind === 'ship' && unit.type === 'cv') return;

  for (const u of units) {
    if (u.hp <= 0) continue;
    if (u.side === unit.side) continue;
    if (u.kind !== 'ship') continue;
    if (manhattan(unit, u) > unit.range) continue;
    if (unit.side === 'player' && !isUnitVisibleToPlayer(u)) continue;
    attackableSet.add(`${u.x},${u.y}`);
  }
}

function refreshLaunchRange(carrier) {
  launchRange.clear();
  if (!carrier || carrier.kind !== 'ship' || carrier.type !== 'cv') return;
  if (carrier.planeCooldown > 0 || carrier.hasPlane) return;
  for (const [dx, dy] of [[1,0],[-1,0],[0,1],[0,-1]]) {
    const nx = carrier.x + dx, ny = carrier.y + dy;
    if (nx < 0 || nx >= SIZE || ny < 0 || ny >= SIZE) continue;
    if (map[ny][nx] === 1) continue;
    if (getUnitAt(nx, ny)) continue;
    launchRange.add(`${nx},${ny}`);
  }
}

function render() {
  boardEl.innerHTML = '';

  for (let y = 0; y < SIZE; y++) {
    for (let x = 0; x < SIZE; x++) {
      const cell = document.createElement('div');
      cell.className = 'cell';
      cell.dataset.x = x;
      cell.dataset.y = y;
      const key = `${x},${y}`;
      const visible = isVisible(x, y);

      if (map[y][x] === 1) cell.classList.add('island');
      if (!visible) cell.classList.add('fog');

      if (visible) {
        if (selected && moveRange.has(key)) cell.classList.add('move-range');
        if (attackableSet.has(key)) cell.classList.add('attack-target');
        if (launchRange.has(key)) cell.classList.add('launch-range');
      }

      const u = getUnitAt(x, y);
      const shouldDraw = u && (u.side === 'player' || visible);

      if (shouldDraw) {
        if (u === selected) cell.classList.add('selected');
        const div = document.createElement('div');
        if (u.kind === 'plane') {
          div.className = `unit plane ${u.side}` + (u.acted ? ' acted' : '');
          div.innerHTML = `<div class="plane-icon">✈</div>`;
        } else {
          div.className = `unit ${u.side}` + (u.acted ? ' acted' : '');
          const pct = Math.max(0, u.hp / u.maxHp * 100);
          let extra = '';
          if (u.type === 'cv') {
            if (u.planeCooldown > 0) extra = `<div style="font-size:9px;color:#ffd54f">整备${u.planeCooldown}</div>`;
            else if (u.hasPlane) extra = `<div style="font-size:9px;color:#4dd0e1">机在空</div>`;
          }
          div.innerHTML = `
            <div class="uname">${u.name}</div>
            <div class="hpbar"><div class="hpfill" style="width:${pct}%;background:${pct > 50 ? '#4caf50' : pct > 25 ? '#ff9800' : '#f44336'}"></div></div>
            ${extra}
          `;
        }
        cell.appendChild(div);
      }

      boardEl.appendChild(cell);
    }
  }

  undoBtn.style.display = (selected && selected.kind === 'ship' && selected.hasMoved) ? 'block' : 'none';
  updateInfo();
  updateEndTurnBtn();
}

boardEl.addEventListener('click', (e) => {
  const cell = e.target.closest('.cell');

function onCellClick(x, y) {
  if (phase !== 'player') return;
  if (isProcessing) return;
  isProcessing = true;
  try { handleClick(x, y); }
  finally { isProcessing = false; }
}

function handleClick(x, y) {
  const key = `${x},${y}`;
  const visible = isVisible(x, y);
  const clickedUnit = getUnitAt(x, y);

  if (!visible) { if (selected) deselect(); return; }

  if (!selected) {
    if (clickedUnit && clickedUnit.side === 'player' && !clickedUnit.acted) selectUnit(clickedUnit);
    return;
  }

  if (clickedUnit && clickedUnit.side === 'enemy' && attackableSet.has(key)) { doAttack(selected, clickedUnit); return; }
  if (selected.kind === 'ship' && selected.type === 'cv' && launchRange.has(key) && !clickedUnit) { launchPlane(selected, x, y); return; }
  if (clickedUnit === selected) { deselect(); return; }
  if (clickedUnit && clickedUnit.side === 'player' && !clickedUnit.acted) { selectUnit(clickedUnit); return; }
  if (moveRange.has(key) && !clickedUnit) { moveUnitTo(selected, x, y); return; }
  if (selected.kind === 'plane') deselect();
  else finishUnit(selected);
}

function moveUnitTo(unit, x, y) {
  if (unit.kind === 'plane') {
    const cost = bfsCost(unit, x, y);
    if (cost > unit.moveLeft) return;
    unit.x = x; unit.y = y;
    unit.moveLeft = Math.max(0, unit.moveLeft - cost);
    unit.justLaunched = false;
    recomputeVision();
    moveRange = getMoveRange(unit);
    refreshAttackable(unit);
    refreshLaunchRange(unit);
    render();
  } else {
    unit.x = x; unit.y = y;
    unit.hasMoved = true;
    recomputeVision();
    moveRange.clear();
    refreshAttackable(unit);
    refreshLaunchRange(unit);
    render();
  }
}

function bfsCost(unit, tx, ty) {
  const visited = new Map();
  visited.set(`${unit.x},${unit.y}`, 0);
  const queue = [{ x: unit.x, y: unit.y, cost: 0 }];
  const isPlane = unit.kind === 'plane';

  while (queue.length > 0) {
    const cur = queue.shift();
    if (cur.x === tx && cur.y === ty) return cur.cost;
    for (const [dx, dy] of [[1,0],[-1,0],[0,1],[0,-1]]) {
      const nx = cur.x + dx, ny = cur.y + dy;
      if (nx < 0 || nx >= SIZE || ny < 0 || ny >= SIZE) continue;
      if (!isPlane && map[ny][nx] === 1) continue;
      const key = `${nx},${ny}`;
      if (visited.has(key)) continue;
      const occ = getUnitAt(nx, ny);
      if (occ && occ !== unit) continue;
      visited.set(key, cur.cost + 1);
      queue.push({ x: nx, y: ny, cost: cur.cost + 1 });
    }
  }
  return 999;
}

function selectUnit(u) {
  selected = u;
  moveRange = getMoveRange(u);
  refreshAttackable(u);
  refreshLaunchRange(u);
  render();
}

function deselect() {
  selected = null;
  moveRange.clear(); attackableSet.clear(); launchRange.clear();
  render();
}

function finishUnit(u) {
  u.acted = true;
  u.hasMoved = false;
  deselect();
  if (checkGameOver()) return;
  if (u.kind === 'plane') { render(); return; }
  const alive = units.filter(x => x.side === 'player' && x.hp > 0 && x.kind === 'ship');
  if (alive.every(x => x.acted)) setTimeout(startEnemyTurn, 700);
}

undoBtn.onclick = () => {
  if (!selected || selected.kind !== 'ship' || !selected.hasMoved) return;
  selected.x = selected.originX;
  selected.y = selected.originY;
  selected.hasMoved = false;
  recomputeVision();
  moveRange = getMoveRange(selected);
  refreshAttackable(selected);
  refreshLaunchRange(selected);
  render();
};

function launchPlane(carrier, x, y) {
  const plane = createPlane(carrier, x, y);
  units.push(plane);
  carrier.hasPlane = true;
  log(`${carrier.name} 放飞舰载机`, 'plane');
  carrier.acted = true;
  carrier.hasMoved = false;
  recomputeVision();
  deselect();
  if (checkGameOver()) return;
  const alive = units.filter(x => x.side === 'player' && x.hp > 0 && x.kind === 'ship');
  if (alive.every(x => x.acted)) setTimeout(startEnemyTurn, 700);
}

/* ================================================================
   攻击
   ================================================================ */

function doAttack(attacker, defender) {
  if (attacker.kind === 'plane') {
    let aa = defender.aa;
    let aaBonus = 0;
    if (defender.type === 'cv') {
      const near = friendlyPlanesNear(defender, 3);
      if (near > 0) { aaBonus = 30; aa += 30; }
    }
    const interceptRate = aa / 100;
    if (Math.random() < interceptRate) {
      log(`${attacker.name} 突防 ${defender.name} —— 被防空火力拦截！(AA ${aa}${aaBonus ? ' 含+30护航' : ''})`, 'aa');
      attacker.hp = 0;
      onPlaneLost(attacker, 'shot');
      return;
    }
    let dmg = attacker.atk;
    const crit = Math.random() < 0.15;
    if (crit) dmg = Math.round(dmg * 1.6);
    defender.hp -= dmg;
    log(`${attacker.name} 突防成功 → ${defender.name}  ${dmg} 伤害${crit ? '（暴击）' : ''}${aaBonus ? ` [敌AA ${aa}]` : ` [AA ${aa}]`}`, 'plane');
    if (defender.hp <= 0) {
      defender.hp = 0;
      log(`  ${defender.name} 被击沉`, 'dmg');
      countKill(defender);
    }
    attacker.hp = 0;
    onPlaneLost(attacker, 'return');
    return;
  }

  if (Math.random() < (defender.dodge || 0)) {
    log(`${attacker.name} → ${defender.name} —— 未命中`, 'dmg');
    finishUnit(attacker);
    return;
  }

  let dmg = attacker.atk * (1 - defender.def / (defender.def + 30));
  const crit = Math.random() < 0.12;
  if (crit) dmg *= 1.6;
  dmg = Math.max(1, Math.round(dmg));
  defender.hp -= dmg;

  if (defender.hp <= 0) {
    defender.hp = 0;
    log(`${attacker.name} → ${defender.name}  ${dmg} 伤害${crit ? '（暴击）' : ''}  【击沉】`, 'dmg');
    countKill(defender);
  } else {
    log(`${attacker.name} → ${defender.name}  ${dmg} 伤害${crit ? '（暴击）' : ''}  剩余 ${defender.hp}`, 'dmg');
  }

  finishUnit(attacker);
}

function countKill(u) {
  if (u.side === 'enemy') battleStats.kills++;
  else battleStats.losses++;
}

function onPlaneLost(plane, reason) {
  const carrier = getCarrier(plane);
  if (carrier && carrier.hp > 0) {
    carrier.hasPlane = false;
    carrier.planeCooldown = 2;
  }
  if (reason === 'return') log(`  ${plane.name} 返航`, 'plane');
  else if (reason === 'crash') log(`  ${plane.name} 未能返航，坠毁`, 'aa');
  else if (reason === 'shot') log(`  ${plane.name} 被击落`, 'aa');

  units = units.filter(u => u.id !== plane.id);
  if (selected && selected.id === plane.id) {
    selected = null; moveRange.clear(); attackableSet.clear(); launchRange.clear();
  }
  recomputeVision();
  render();
  if (checkGameOver()) return;
  if (phase === 'player') {
    const alive = units.filter(x => x.side === 'player' && x.hp > 0 && x.kind === 'ship');
    if (alive.length > 0 && alive.every(x => x.acted)) setTimeout(startEnemyTurn, 700);
  }
}

/* ================================================================
   敌方回合
   ================================================================ */

function startEnemyTurn() {
  if (phase === 'over') return;
  phase = 'enemy';
  selected = null;
  moveRange.clear(); attackableSet.clear(); launchRange.clear();
  render();
  log(`— 敌方回合 —`, 'turn');

  const enemyShips = units.filter(u => u.side === 'enemy' && u.hp > 0 && u.kind === 'ship');
  const enemyPlanes = units.filter(u => u.side === 'enemy' && u.hp > 0 && u.kind === 'plane');
  const sequence = [...enemyShips, ...enemyPlanes];
  let i = 0;

  function next() {
    if (checkGameOver()) return;
    if (i >= sequence.length) { endEnemyTurn(); return; }
    const e = sequence[i++];
    if (e.hp <= 0) { next(); return; }
    doEnemyAction(e);
    recomputeVision();
    render();
    if (checkGameOver()) return;
    setTimeout(next, 600);
  }
  setTimeout(next, 400);
}

function doEnemyAction(e) {
  const playerShips = units.filter(u => u.side === 'player' && u.hp > 0 && u.kind === 'ship');
  if (playerShips.length === 0) return;

  if (e.kind === 'plane') {
    let target = null, bestD = Infinity;
    for (const t of playerShips) {
      const d = manhattan(e, t);
      if (d < bestD) { bestD = d; target = t; }
    }
    if (!target) return;
    if (manhattan(e, target) <= e.range) { doAttack(e, target); return; }

    const range = getMoveRange(e);
    let bestPos = null, bestDist = manhattan(e, target);
    for (const key of range) {
      const [x, y] = key.split(',').map(Number);
      const d = Math.abs(x - target.x) + Math.abs(y - target.y);
      if (d < bestDist) { bestDist = d; bestPos = { x, y }; }
    }
    if (bestPos) {
      e.x = bestPos.x; e.y = bestPos.y;
      if (manhattan(e, target) <= e.range) doAttack(e, target);
    }
    return;
  }

  if (e.type === 'cv' && e.planeCooldown === 0 && !e.hasPlane) {
    const spots = [];
    for (const [dx, dy] of [[1,0],[-1,0],[0,1],[0,-1]]) {
      const nx = e.x + dx, ny = e.y + dy;
      if (nx < 0 || nx >= SIZE || ny < 0 || ny >= SIZE) continue;
      if (map[ny][nx] === 1) continue;
      if (getUnitAt(nx, ny)) continue;
      spots.push({ x: nx, y: ny });
    }
    if (spots.length > 0) {
      const spot = spots[Math.floor(Math.random() * spots.length)];
      units.push(createPlane(e, spot.x, spot.y));
      e.hasPlane = true;
      log(`${e.name} 放飞舰载机`, 'plane');
      return;
    }
  }
  if (e.type === 'cv') return;

  let target = playerShips[0], bestScore = Infinity;
  for (const t of playerShips) {
    const score = manhattan(e, t) * 10 + (t.hp / t.maxHp) * 6;
    if (score < bestScore) { bestScore = score; target = t; }
  }
  if (manhattan(e, target) <= e.range) { doAttack(e, target); return; }

  const range = getMoveRange(e);
  let bestPos = null, bestDist = manhattan(e, target);
  for (const key of range) {
    const [x, y] = key.split(',').map(Number);
    const d = Math.abs(x - target.x) + Math.abs(y - target.y);
    if (d < bestDist) { bestDist = d; bestPos = { x, y }; }
  }
  if (bestPos) {
    e.x = bestPos.x; e.y = bestPos.y;
    if (manhattan(e, target) <= e.range) doAttack(e, target);
  }
}

function endEnemyTurn() {
  if (checkGameOver()) return;

  for (const u of units) {
    if (u.kind === 'ship' && u.type === 'cv' && u.planeCooldown > 0) u.planeCooldown--;
  }

  for (const u of units) {
    if (u.side === 'player') {
      if (u.kind === 'ship') {
        u.acted = false; u.hasMoved = false;
        u.originX = u.x; u.originY = u.y;
      } else if (u.kind === 'plane') {
        u.acted = false; u.hasMoved = false;
        u.moveLeft = u.move;
        u.originX = u.x; u.originY = u.y;
      }
    }
  }

  const planesToRemove = [];
  for (const u of units) {
    if (u.kind !== 'plane' || u.hp <= 0) continue;
    if (u.justLaunched) { u.justLaunched = false; continue; }
    u.lifeLeft--;
    if (u.lifeLeft <= 0) planesToRemove.push(u);
  }
  for (const p of planesToRemove) {
    log(`${p.name} 燃油耗尽，坠毁`, 'aa');
    p.hp = 0;
    onPlaneLost(p, 'crash');
  }

  turnCount++;
  battleStats.turns = turnCount - 1;
  log(`— 第 ${turnCount} 回合 —`, 'turn');
  phase = 'player';
  recomputeVision();
  render();
}

/* ================================================================
   胜负
   ================================================================ */

function checkGameOver() {
  const pAlive = units.some(u => u.side === 'player' && u.hp > 0 && u.kind === 'ship');
  const eAlive = units.some(u => u.side === 'enemy' && u.hp > 0 && u.kind === 'ship');

  if (!pAlive) { endBattle('defeat'); return true; }
  if (!eAlive) { endBattle('victory'); return true; }
  return false;
}

function endBattle(result) {
  phase = 'over';
  battleStats.turns = turnCount;

  let newlyCleared = false;
  if (result === 'victory' && currentLevel) {
    if (!isCleared(currentLevel.id)) newlyCleared = true;
    markCleared(currentLevel.id);
  }

  const title = document.getElementById('resultTitle');
  title.textContent = result === 'victory' ? '胜 利 ！' : '败 北 ...';
  title.className = 'result-title ' + result;

  const sub = document.getElementById('resultSub');
  if (result === 'victory') {
    const nextLevel = LEVELS.find(l => l.id === currentLevel.id + 1);
    if (nextLevel) {
      sub.textContent = newlyCleared
        ? `已解锁：第 ${nextLevel.id} 关 · ${nextLevel.name}`
        : `下一关：第 ${nextLevel.id} 关 · ${nextLevel.name}`;
    } else {
      sub.textContent = '全部通关！';
    }
  } else {
    sub.textContent = '调整舰队配置，再试一次';
  }

  // ★ 结算立绘：选一个己方还活着的舰娘当 MVP
  const resultPortraitEl = document.getElementById('resultPortrait');
  resultPortraitEl.classList.add('hidden');
  const survivors = units.filter(u => u.side === 'player' && u.hp > 0 && u.kind === 'ship' && u.portrait);
  if (survivors.length > 0) {
    // 取第一个有立绘的（或按某种规则，比如剩余 HP 最高的）
    survivors.sort((a, b) => b.hp - a.hp);
    const mvp = survivors[0];
    resultPortraitEl.src = mvp.portrait;
    resultPortraitEl.alt = mvp.name;
    resultPortraitEl.classList.remove('hidden');
    resultPortraitEl.onerror = () => resultPortraitEl.classList.add('hidden');
  }

  document.getElementById('statTurns').textContent = battleStats.turns;
  document.getElementById('statKills').textContent = battleStats.kills;
  document.getElementById('statLosses').textContent = battleStats.losses;

  const actionsEl = document.getElementById('resultActions');
  actionsEl.innerHTML = '';

  if (result === 'victory') {
    const nextLevel = LEVELS.find(l => l.id === currentLevel.id + 1);
    if (nextLevel) {
      const btn = document.createElement('button');
      btn.className = 'primary';
      btn.textContent = '下 一 关';
      btn.onclick = () => startLevel(nextLevel.id);
      actionsEl.appendChild(btn);
    }
    const backBtn = document.createElement('button');
    backBtn.textContent = '返回关卡';
    backBtn.onclick = () => gotoCampaign();
    actionsEl.appendChild(backBtn);
  } else {
    const retryBtn = document.createElement('button');
    retryBtn.className = 'primary';
    retryBtn.textContent = '重 试';
    retryBtn.onclick = () => startLevel(currentLevel.id);
    actionsEl.appendChild(retryBtn);

    const backBtn = document.createElement('button');
    backBtn.textContent = '返回关卡';
    backBtn.onclick = () => gotoCampaign();
    actionsEl.appendChild(backBtn);
  }

  setTimeout(() => showScreen('screen-result'), 800);
}
